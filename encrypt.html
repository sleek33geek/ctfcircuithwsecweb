Encrypt Module (SystemVerilog)

Below is the hardware implementation of the 8-block parallel encrypter:


module encrypt (
    input  logic        clk,
    input  logic        rst,
    input  logic        en,
    input  logic [63:0] plaintext,  // message for encryption (64 bits)
    input  logic [7:0]  key,        // Cryptographic key

    output logic [63:0] ciphertext  // encrypted message (64 bits)
);

    genvar i;

// 8 Identical Modules of Encryption
    generate
        for (i = 0; i < 8; i++) begin : GEN_ENCRYPT_BLOCK

            // Signal Definition
            logic [1:0] k2k1;

            logic en_AES_i;
            logic en_Scrambler_i;
            logic en_otherthing_i;

            logic [7:0] out_AES_i;
            logic [7:0] out_Scrambler_i;
            logic [7:0] out_otherthing_i;


            // Local 2-bit key
            if (i == 7) begin
                assign k2k1 = {key[0], key[7]};
            end
            else begin
                assign k2k1 = {key[i+1], key[i]};
            end


            // Encryption block enables
            always_comb begin
                // default Assignment
                en_AES_i        = 1'b0;
                en_Scrambler_i  = 1'b0;
                en_otherthing_i = 1'b0;
                
                // en - block activation assignment
                case (k2k1)
                    2'b00: en_AES_i        = en;
                    2'b01: en_Scrambler_i  = en;
                    2'b10: en_otherthing_i = en;
                    default: begin
                     // 2'b11 -> no encryption block enabled
                    end
                endcase
            end


            // Encryption modules
            /*Start of Encryption module instantiation section*/

            // AES
            enc_AES u_enc_AES (
                .clk     (clk),
                .rst     (rst),
                .en      (en_AES_i),
                .ptxt    (plaintext[i*8 +: 8]),
                .key     (k2k1),
                .cphrtxt (out_AES_i)
            );
            
            // Scrambler
            enc_Scrambler u_enc_Scrambler (
                .clk     (clk),
                .rst     (rst),
                .en      (en_Scrambler_i),
                .ptxt    (plaintext[i*8 +: 8]),
                .key     (k2k1),
                .cphrtxt (out_Scrambler_i)
            );

            //otherthing
            enc_otherthing u_enc_otherthing (
                .clk     (clk),
                .rst     (rst),
                .en      (en_otherthing_i),
                .ptxt    (plaintext[i*8 +: 8]),
                .key     (k2k1),
                .cphrtxt (out_otherthing_i)
            );
            /*End of Encryption module instantiation section*/


            // Ciphertext Composition
            always_comb begin
                case (k2k1)
                    2'b00:    ciphertext[i*8 +: 8] = out_AES_i;
                    2'b01:    ciphertext[i*8 +: 8] = out_Scrambler_i;
                    2'b10:    ciphertext[i*8 +: 8] = out_otherthing_i;
                    default:  ciphertext[i*8 +: 8] = 8'b0;
                endcase
            end

        end
    endgenerate

endmodule
