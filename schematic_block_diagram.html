encrypt.sv (Verilog Implementation) SystemVerilog Multi-Cipher Block 
module encrypt (
    input  logic         clk,
    input  logic         rst,
    input  logic         en,
    input  logic [1023:0] plaintext,  // message for encryption (64 bits)
    input  logic [1023:0] key,       // Cryptographic key
 
    output logic [1023:0] ciphertext  // encrypted message (64 bits)
);
 
    genvar i;
 
    logic [15:0] enc_key = key [15:0];
 
// 8 Identical Modules of Encryption
    generate
        for (i = 0; i < 8; i++) begin : GEN_ENCRYPT_BLOCK
 
            // Signal Definition
            logic [1:0] k2k1;
 
            logic en_AES;
            logic en_Twofish;
            logic en_Skinny;
            logic en_Midori;
 
            logic [127:0] out_AES;
            logic [127:0] out_Twofish;
            logic [127:0] out_Skinny;
            logic [127:0] out_Midori;
 
            // Local 2-bit key
            assign k2k1 = {key[2*i+1], key[2*i]};
 
            // Encryption block enables
            always_comb begin
                // default Assignment
                en_AES     = 1'b0;
                en_Twofish = 1'b0;
                en_Skinny  = 1'b0;
                en_Midori  = 1'b0;
                
                // en - block activation assignment
                case (k2k1)
                    2'b00:   en_AES     = en;
                    2'b01:   en_Twofish = en;
                    2'b10:   en_Skinny  = en;
                    2'b11:   en_Midori  = en;
                    default: ;
                endcase
            end
 
            // Encryption modules
            /*Start of Encryption module instantiation section*/
 
            // AES
            enc_AES u_enc_AES (
                .clk     (clk),
                .rst     (rst),
                .en      (en_AES),
                .ptxt    (plaintext[i*128 +: 128]),
                .key     (k2k1),
                .cphrtxt (out_AES)
            );
            
            // Twofish
            enc_Twofish u_enc_Twofish (
                .clk     (clk),
                .rst     (rst),
                .en      (en_Twofish),
                .ptxt    (plaintext[i*128 +: 128]),
                .key     (k2k1),
                .cphrtxt (out_Twofish)
            );
 
            // Skinny
            enc_Skinny u_enc_Skinny (
                .clk     (clk),
                .rst     (rst),
                .en      (en_Skinny),
                .ptxt    (plaintext[i*128 +: 128]),
                .key     (k2k1),
                .cphrtxt (out_Skinny)
            );
 
            // Midori
            enc_Midori u_enc_Midori (
                .clk     (clk),
                .rst     (rst),
                .en      (en_Midori),
                .ptxt    (plaintext[i*128 +: 128]),
                .key     (k2k1),
                .cphrtxt (out_Midori)
            );
 
            /*End of Encryption module instantiation section*/
 
            // Ciphertext Composition
            always_comb begin
                case (k2k1)
                    2'b00:   ciphertext[i*128 +: 128] = out_AES;
                    2'b01:   ciphertext[i*128 +: 128] = out_Twofish;
                    2'b10:   ciphertext[i*128 +: 128] = out_Skinny;
                    2'b11:   ciphertext[i*128 +: 128] = out_Midori;
 
                    default: ciphertext[i*128 +: 128] = 128'b0;
                endcase
            end
 
        end
    endgenerate
 
endmodule
